    71→

   124→        const c = cfg as {

   144→    const onEnded = () => setPlaying(false);
   145→    const onLoaded = () => setSourceReady(true);
   146→    const onErr = () => setSourceReady(false);

   148→    a.addEventListener("pause", onPause);
   149→    a.addEventListener("ended", onEnded);
   150→    a.addEventListener("canplay", onLoaded);
   151→    a.addEventListener("loadedmetadata", onLoaded);
   152→    a.addEventListener("error", onErr);

   154→      a.removeEventListener("play", onPlay);
   155→      a.removeEventListener("pause", onPause);
   156→      a.removeEventListener("ended", onEnded);
   157→      a.removeEventListener("canplay", onLoaded);
   158→      a.removeEventListener("loadedmetadata", onLoaded);
   159→      a.removeEventListener("error", onErr);
   160→    };

   192→      const dur = useProjectStore.getState().analysis?.durationSec ?? 0;

   194→      const a = audioRef.current;
   195→      if (a && a.src) {
   196→        try {
   197→          a.currentTime = tt;
   198→        } catch {
   199→          /* seek antes de metadata */

   202→      useProjectStore.getState().setPlayhead(tt);
   203→    },
   204→    []
   205→  );
   206→

   208→    const a = audioRef.current;
   209→    if (!a || !a.src) return;
   210→    if (a.paused) {
   211→      a.play().catch(() => undefined);
   212→    } else {
   213→      a.pause();
   214→    }
   215→  }, []);

   218→    const a = audioRef.current;
   219→    if (a) {
   220→      a.pause();
   221→      try {
   222→        a.currentTime = 0;
   223→      } catch {
   224→        /* noop */
   225→      }

   233→      if (e.code !== "Space") return;
   234→      const target = e.target as HTMLElement | null;
   235→      if (
   236→        target &&
   237→        (["BUTTON", "INPUT", "TEXTAREA", "SELECT"].includes(target.tagName) ||
   238→          target.getAttribute("role") === "slider" ||
   239→          target.getAttribute("role") === "checkbox")
   240→      )
   241→        return;
   242→      e.preventDefault();
   243→      togglePlay();
   244→    };
   245→    window.addEventListener("keydown", onKey);
   246→    return () => window.removeEventListener("keydown", onKey);
   247→  }, [togglePlay]);

   261→          cancel: () => cancelRef.current,
   262→        });

   264→      } catch (err) {
   265→        const msg = err instanceof Error ? err.message : String(err);
   266→        if (msg.includes("cancel")) {
   267→          useProjectStore.getState().cancelAnalysis();
   268→        } else {
   269→          useProjectStore.getState().failAnalysis(msg);
   270→        }
   271→      }
   272→    },
   273→    []

   279→    objectUrlRef.current = url;

   281→    void kind;
   282→  };

   290→        name: file.name,
   291→        size: file.size,
   292→        type: file.type || "audio/*",
   293→        lastModified: file.lastModified || Date.now(),
   294→        key,
   295→      };
   296→      void runAnalysis(file, meta, "user");
   297→    },
   298→    [runAnalysis]

   312→      await runAnalysis(file, { name: file.name, size: file.size, type: "audio/wav", lastModified: 0, key }, "demo");

   314→      useProjectStore.getState().failAnalysis("No se pudo descargar la demo.");

   316→      setDemoLoading(false);
   317→    }

   326→    const key = useProjectStore.getState().fileMeta?.key;

   340→  // ---------- Fase 2: transcripción + tensión ----------

   355→        blob = await (await fetch(url)).blob();

   368→            hopSec: a.envelopeHopSec,
   369→            durationSec: a.durationSec,
   370→            silencios: st.silences,
   371→            umbralDb: st.thresholds.dbThreshold,
   372→            config,
   373→            claveFuente: st.fileMeta.key,
   374→            previos: rehacer

   377→                  segmentos: st.fase2.segmentos,
   378→                  bloquesHechos: st.fase2.bloquesHechos,

   382→            onProgreso: (p) => useProjectStore.getState().setProgresoFase2(p),

   392→          .getState()

   397→          useProjectStore.getState().cancelarFase2();
   398→        } else {
   399→          useProjectStore.getState().fallarFase2(msg);

   453→

   456→  const fase2Listo =

   479→      {/* ---------------- cabecera ---------------- */}

   481→        <div className="mx-auto flex w-full max-w-6xl items-center gap-3 px-4 py-3 sm:px-6 lg:px-8">
   482→          <span className="flex h-9 w-9 items-center justify-center rounded-lg bg-amber-500/15 text-amber-400">

   514→                ? "review"
   515→                : fase2.status
   516→              : "locked"
   517→          }

   520→        <AnimatePresence mode="wait">

   526→              exit={{ opacity: 0, y: -8 }}
   527→              className="space-y-5"
   528→            >
   529→              <div className="space-y-2 pt-2">
   530→                <h2 className="text-xl font-bold sm:text-2xl">
   531→                  Fase 1 · Ingestión y análisis visual
   532→                </h2>
   533→                <p className="max-w-3xl text-sm leading-relaxed text-zinc-400">
   534→                  Carga la narración de tu sesión: verás su{" "}
   535→                  <strong className="text-zinc-200">forma de onda</strong>, los{" "}
   536→                  <strong className="text-rose-300">silencios</strong> (pausas

   538→                  umbral antes de aprobar. El audio se procesa{" "}

   540→                    por bloques en tu propio navegador
   541→                  </strong>{" "}
