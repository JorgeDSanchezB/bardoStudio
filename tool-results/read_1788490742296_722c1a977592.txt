   298→const grow = (arr: Float32Array, len: number, fill: number): Float32Array => {
   299→  if (arr.length >= len) return arr;
   300→  const out = new Float32Array(len);
   301→  out.set(arr);
   302→  if (fill !== 0) out.fill(fill, arr.length);
   303→  return out;
   304→};

   335→function normalizarConfigFase3(bruto: unknown): Fase3Config {
   336→  const d = FASE3_CONFIG_DEFECTO;
   337→  if (!bruto || typeof bruto !== "object") return d;
   338→  const c = bruto as Partial<Fase3Config>;
   339→  const num = (v: unknown, def: number, min: number, max: number) => {
   340→    const n = Number(v);
   341→    return Number.isFinite(n) ? Math.min(max, Math.max(min, n)) : def;
   342→  };
   343→  return {
   344→    duracionMinBloqueSeg: num(
   345→      c.duracionMinBloqueSeg,
   346→      d.duracionMinBloqueSeg,
   347→      5,
   348→      600
   349→    ),
   350→    penalizacionFatigaMin: num(
   351→      c.penalizacionFatigaMin,
   352→      d.penalizacionFatigaMin,
   353→      1,
   354→      60
   355→    ),
   356→  };
   357→}

   442→export const useProjectStore = create<ProjectStore>()(
   443→  persist(
   444→    (set, get) => ({
   445→      ...initial,
   446→
   447→      seedFromConfig: (cfg) => {
   448→        const s = get();
   449→        const nextOpts: AnalysisOptions = {
   450→          ...s.analysisOpts,
   451→          ...(cfg.analisis ?? {}),
   452→        };
   453→        set({ analysisOpts: nextOpts });
   454→        if (cfg.fase2 !== undefined) {
   455→          set({ fase2: { ...get().fase2, configFase2: normalizarConfigFase2(cfg.fase2) } });
   456→        }
   457→        if (cfg.fase3 !== undefined) {
   458→          set({ configFase3: normalizarConfigFase3(cfg.fase3) });
   459→        }
   460→        if (cfg.fase4 !== undefined) {
   461→          const d = normalizarParamsFase4(cfg.fase4);
   462→          set({
   463→            fase4: {
   464→              ...get().fase4,
   465→              configDefaults: d,
   466→              params: get().fase4.paramsTouched ? get().fase4.params : d,
   467→            },
   468→          });
   469→        }
   470→        if (cfg.fase6 !== undefined) {
   471→          const d = normalizarParamsFase6(cfg.fase6);
   472→          set({
   473→            fase6: {
   474→              ...get().fase6,
   475→              configDefaults: d,
   476→              params: get().fase6.paramsTouched ? get().fase6.params : d,
   477→            },
   478→          });
   479→        }
   480→        const sil = cfg.silencio ?? {};
   481→        const defaults: Thresholds = {
   482→          dbThreshold: sil.umbralDb ?? s.thresholds.dbThreshold,
   483→          minDurationSec: sil.duracionMinimaSeg ?? s.thresholds.minDurationSec,
   484→          mergeGapSec: sil.fusionHuecosSeg ?? s.thresholds.mergeGapSec,
   485→        };
   486→        set({ configDefaults: defaults });
   487→        if (!s.thresholdsTouched) {
   488→          set({ thresholds: defaults });
   489→          get().recomputeSilences();
   490→        }
   491→      },
   492→
   493→      beginAnalysis: (meta, kind) =>
   494→        set({
   495→          fileMeta: meta,
   496→          sourceKind: kind,
   497→          status: "analyzing",
   498→          progress: null,
   499→          analysis: null,
   500→          gridInfo: null,
   501→          silences: [],
   502→          silenceStats: emptyStats,
   503→          savedResult: null,
   504→          analyzedAt: null,
   505→          playheadSec: 0,
   506→          error: null,
   507→          sourceMissing: false,
   508→        }),
   509→
   510→      setMetaInfo: (meta) => {
   511→        const opts = get().analysisOpts;
   512→        const grid = makeGrid(meta.sampleRate, opts);
   513→        set({
   514→          gridInfo: { peaksPerSec: grid.perSec, envelopeHopSec: grid.hopSec },
   515→          analysis: {
   516→            sampleRate: meta.sampleRate,
   517→            channels: meta.channels,
   518→            format: meta.format,
   519→            peaksPerSec: grid.perSec,
   520→            envelopeHopSec: grid.hopSec,
   521→            peaksMin: new Float32Array(0),
   522→            peaksMax: new Float32Array(0),
   523→            envDb: new Float32Array(0),
   524→            durationSec: 0,
   525→          },
   526→        });
   527→      },
   528→
   529→      setProgress: (p) => set({ progress: p }),
   530→
   531→      appendChunk: (c) => {
   532→        const a = get().analysis;
   533→        if (!a || c.nBuckets === 0) {
   534→          if (a && c.nBuckets === 0) {
   535→            set({ analysis: { ...a, durationSec: (c.startSample + c.sampleCount) / a.sampleRate } });
   536→          }
   537→          return;
   538→        }
   539→        const needB = c.startBucket + c.nBuckets;
   540→        const needW = c.startWindow + c.nWindows;
   541→        const peaksMin = grow(a.peaksMin, needB, 1);
   542→        const peaksMax = grow(a.peaksMax, needB, -1);
   543→        const envDb = grow(a.envDb, needW, -100);
   544→        peaksMin.set(c.peaksMin, c.startBucket);
   545→        peaksMax.set(c.peaksMax, c.startBucket);
   546→        envDb.set(c.envDb, c.startWindow);
   547→        set({
   548→          analysis: {
   549→            ...a,
   550→            peaksMin,
   551→            peaksMax,
   552→            envDb,
   553→            durationSec: (c.startSample + c.sampleCount) / a.sampleRate,
   554→          },
   555→        });
   556→      },
   557→
