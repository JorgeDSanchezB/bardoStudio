     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     4→import { Button } from "@/components/ui/button";
     5→import { ZoomIn, ZoomOut, Maximize2 } from "lucide-react";
     6→import type { AnalysisData } from "@/store/project";
     7→import type { SilenceRegion } from "@/lib/audio/silence";
     8→import type { BloqueMusica } from "@/lib/fase4/tipos";
     9→import type { Pista } from "@/lib/fase3/tipos";
    10→import {
    11→  buclesDe,
    12→  limitesBorde,
    13→  type LadoRecorte,
    14→} from "@/lib/fase5/ediciones";
    15→import { ANIMO_COLORES } from "@/lib/fase2/types";
    16→import {
    17→  clamp,
    18→  formatRulerTick,
    19→  formatTimecode,
    20→  pickRulerStep,
    21→} from "@/lib/format";
    22→
    23→interface MixTimelineProps {
    24→  analysis: AnalysisData | null;
    25→  durationSec: number;
    26→  bloques: BloqueMusica[];
    27→  pistas: Pista[];
    28→  nivelDb: Float32Array | null;
    29→  hopSec: number;
    30→  duckDb?: Float32Array | null;
    31→  silences: SilenceRegion[];
    32→  playheadSec: number;
    33→  editable?: boolean;
    34→  selectedId?: string | null;
    35→  onSelect?: (id: string | null) => void;
    36→  onMover?: (id: string, nuevoInicio: number) => void;
    37→  /** Recorte/alargado por los bordes (estilo Camtasia). */
    38→  onRecortar?: (id: string, lado: LadoRecorte, valor: number) => void;
    39→  onSeek: (t: number) => void;
    40→}
    41→
    42→const RULER_H = 26;
    43→const NARR_H = 64;
    44→const MUS_H = 132;
    45→const MAX_PPS = 2000;
    46→/** ancho en px de la zona de agarre de cada borde de bloque */
    47→const EDGE_PX = 9;
    48→/** rango vertical del carril de música: dB → y */
    49→const DB_MIN = -48;
    50→const DB_MAX = 6;
    51→
    52→type View = { start: number; pps: number };
    53→
    54→/** Arrastre de cuerpo: desplaza el bloque completo. */
    55→interface ArrastreMover {
    56→  modo: "mover";
    57→  id: string;
    58→  inicioOriginal: number;
    59→  deltaSec: number;
    60→}
    61→
    62→/** Arrastre de borde: recorta (hacia dentro) o alarga (hacia fuera). */
    63→interface ArrastreRecorte {
    64→  modo: "recorte";
    65→  id: string;
    66→  lado: LadoRecorte;
    67→  valorOriginal: number;
    68→  valor: number;
    69→}
    70→
    71→type Arrastre = ArrastreMover | ArrastreRecorte;
    72→
    73→export function MixTimeline(props: MixTimelineProps) {
    74→  const {
    75→    analysis,
    76→    durationSec,
    77→    bloques,
    78→    pistas,
    79→    nivelDb,
    80→    hopSec,
    81→    duckDb,
    82→    silences,
    83→    playheadSec,
    84→    editable,
    85→    selectedId,
    86→    onSelect,
    87→    onMover,
    88→    onRecortar,
    89→    onSeek,
    90→  } = props;
    91→
    92→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
    93→  const wrapRef = useRef<HTMLDivElement | null>(null);
    94→  const [ancho, setAncho] = useState(900);
    95→  const [vista, setVista] = useState<View | null>(null);
    96→  const [arrastre, setArrastre] = useState<Arrastre | null>(null);
    97→  const [hover, setHover] = useState<{ x: number; t: number } | null>(null);
    98→
    99→  const pistaMap = useMemo(
   100→    () => new Map(pistas.map((p) => [p.id, p])),
   101→    [pistas]
   102→  );
   103→
   104→  /** Aplica el arrastre en curso a un bloque (vista previa en vivo). */
   105→  const conArrastre = useCallback(
   106→    (b: BloqueMusica): BloqueMusica => {
   107→      if (!arrastre || arrastre.id !== b.id) return b;
   108→      if (arrastre.modo === "mover") {
   109→        return {
   110→          ...b,
   111→          inicio: b.inicio + arrastre.deltaSec,
   112→          fin: b.fin + arrastre.deltaSec,
   113→        };
   114→      }
   115→      const lim = limitesBorde(b, durationSec);
   116→      if (arrastre.lado === "inicio") {
   117→        const v = clamp(
   118→          Math.round(arrastre.valor * 10) / 10,
   119→          lim.inicioMin,
   120→          lim.inicioMax
   121→        );
   122→        const dur = b.fin - v;
   123→        return {
   124→          ...b,
   125→          inicio: v,
   126→          offsetOrigen: Math.max(0, (b.offsetOrigen ?? 0) + (v - b.inicio)),
   127→          fadeInSeg: Math.min(b.fadeInSeg, dur / 2),
   128→          fadeOutSeg: Math.min(b.fadeOutSeg, dur / 2),
   129→        };
   130→      }
   131→      const v = clamp(
   132→        Math.round(arrastre.valor * 10) / 10,
   133→        lim.finMin,
   134→        lim.finMax
   135→      );
   136→      const dur = v - b.inicio;
   137→      return {
   138→        ...b,
   139→        fin: v,
   140→        fadeInSeg: Math.min(b.fadeInSeg, dur / 2),
   141→        fadeOutSeg: Math.min(b.fadeOutSeg, dur / 2),
   142→      };
   143→    },
   144→    [arrastre, durationSec]
   145→  );
   146→
   147→  const ponerCursor = (css: string) => {
   148→    const el = canvasRef.current;
   149→    if (el && el.style.cursor !== css) el.style.cursor = css;
   150→  };
   151→
   152→  // ---------- tamaño ----------
   153→  useEffect(() => {
   154→    const el = wrapRef.current;
   155→    if (!el) return;
   156→    const ro = new ResizeObserver(() => setAncho(Math.max(320, el.clientWidth)));
   157→    ro.observe(el);
   158→    setAncho(Math.max(320, el.clientWidth));
   159→    return () => ro.disconnect();
   160→  }, []);
   161→
   162→  // ---------- vista efectiva (derivada en render: fit por defecto) ----------
   163→  const vistaEfectiva = useMemo(
   164→    () => {
   165→      const base =
   166→        vista ?? {
   167→          start: 0,
   168→          pps: clamp(ancho / Math.max(1, durationSec), 0.05, MAX_PPS),
   169→        };
   170→      const pps = clamp(base.pps, 0.05, MAX_PPS);
   171→      const maxStart = Math.max(0, durationSec - ancho / pps);
   172→      return { start: clamp(base.start, 0, maxStart), pps };
   173→    },
   174→    [vista, ancho, durationSec]
   175→  );
   176→
   177→  const vistaParaDibujo = arrastre ? { ...vistaEfectiva } : vistaEfectiva;
   178→
   179→  // ---------- dibujo ----------
   180→  const dibujar = useCallback(() => {
   181→    const canvas = canvasRef.current;
   182→    if (!canvas) return;
   183→    const dpr = Math.min(2, window.devicePixelRatio || 1);
   184→    const W = ancho;
   185→    const H = RULER_H + NARR_H + MUS_H;
   186→    if (canvas.width !== W * dpr || canvas.height !== H * dpr) {
   187→      canvas.width = W * dpr;
   188→      canvas.height = H * dpr;
   189→    }
   190→    const ctx = canvas.getContext("2d");
   191→    if (!ctx) return;
   192→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   193→    ctx.clearRect(0, 0, W, H);
   194→
   195→    const { start: t0, pps } = vistaParaDibujo;
   196→    const t1 = Math.min(durationSec, t0 + W / pps);
   197→    const xDe = (t: number) => (t - t0) * pps;
   198→
   199→    // ===== regla =====
   200→    ctx.fillStyle = "#18181b";
   201→    ctx.fillRect(0, 0, W, RULER_H);
   202→    ctx.strokeStyle = "#27272a";
   203→    ctx.beginPath();
   204→    ctx.moveTo(0, RULER_H - 0.5);
   205→    ctx.lineTo(W, RULER_H - 0.5);
   206→    ctx.stroke();
   207→    const step = pickRulerStep(pps);
   208→    ctx.font = "10px ui-monospace, SFMono-Regular, Menlo, monospace";
   209→    ctx.textBaseline = "top";
   210→    const firstTick = Math.floor(t0 / step) * step;
   211→    for (let t = firstTick; t <= t1 + step; t += step) {
   212→      if (t < 0) continue;
   213→      const x = Math.round(xDe(t)) + 0.5;
   214→      if (x < -2 || x > W + 2) continue;
   215→      ctx.strokeStyle = "#3f3f46";
   216→      ctx.beginPath();
   217→      ctx.moveTo(x, RULER_H - 10);
   218→      ctx.lineTo(x, RULER_H - 1);
   219→      ctx.stroke();
   220→      ctx.fillStyle = "#a1a1aa";
   221→      ctx.fillText(formatRulerTick(t, step), x + 3, 3);
   222→    }
   223→    const xEnd = xDe(durationSec);
   224→    if (xEnd >= 0 && xEnd <= W) {
   225→      ctx.strokeStyle = "#f43f5e";
   226→      ctx.beginPath();
   227→      ctx.moveTo(xEnd + 0.5, 0);
   228→      ctx.lineTo(xEnd + 0.5, H - 1);
   229→      ctx.stroke();
   230→    }
   231→
   232→    // ===== carril narración =====
   233→    const narrTop = RULER_H;
   234→    ctx.fillStyle = "#09090b";
   235→    ctx.fillRect(0, narrTop, W, NARR_H);
   236→    // etiqueta
   237→    ctx.fillStyle = "#52525b";
   238→    ctx.font = "10px ui-monospace, SFMono-Regular, Menlo, monospace";
   239→    ctx.fillText("VOZ", 6, narrTop + 6);
   240→    if (analysis && analysis.peaksMax.length > 0) {
   241→      const mid = narrTop + NARR_H / 2;
   242→      const amp = (NARR_H / 2 - 6) / Math.max(0.001, analysis.peaksMax.reduce((m, v) => Math.max(m, Math.abs(v)), 0.001));
   243→      const i0 = Math.max(0, Math.floor(t0 * analysis.peaksPerSec));
   244→      const i1 = Math.min(
   245→        analysis.peaksMax.length - 1,
   246→        Math.ceil(t1 * analysis.peaksPerSec)
   247→      );
   248→      ctx.strokeStyle = "rgba(245,158,11,0.75)";
   249→      ctx.lineWidth = 1;
   250→      ctx.beginPath();
   251→      const pxPerPeak = pps / analysis.peaksPerSec;
   252→      if (pxPerPeak >= 1.5) {
   253→        for (let i = i0; i <= i1; i++) {
   254→          const x = xDe(i / analysis.peaksPerSec);
   255→          const mx = mid - analysis.peaksMax[i] * amp;
   256→          const mn = mid - analysis.peaksMin[i] * amp;
   257→          ctx.moveTo(x, mn);
   258→          ctx.lineTo(x, mx);
   259→        }
   260→        ctx.stroke();
   261→      } else {
   262→        // agrupar por píxel
   263→        for (let x = 0; x < W; x++) {
   264→          const ta = t0 + x / pps;
   265→          const tb = ta + 1 / pps;
   266→          const ja = Math.max(i0, Math.floor(ta * analysis.peaksPerSec));
   267→          const jb = Math.min(i1, Math.ceil(tb * analysis.peaksPerSec));
   268→          if (jb < ja) continue;
   269→          let mx = 0;
   270→          let mn = 0;
   271→          for (let j = ja; j <= jb; j++) {
   272→            if (analysis.peaksMax[j] > mx) mx = analysis.peaksMax[j];
   273→            if (analysis.peaksMin[j] < mn) mn = analysis.peaksMin[j];
   274→          }
   275→          ctx.moveTo(x + 0.5, mid - mn * amp);
   276→          ctx.lineTo(x + 0.5, mid - mx * amp);
   277→        }
   278→        ctx.stroke();
   279→      }
   280→    }
   281→    // silencios
   282→    for (const s of silences) {
   283→      if (s.end < t0 || s.start > t1) continue;
   284→      const x = xDe(s.start);
   285→      const x2 = xDe(s.end);
   286→      ctx.fillStyle = "rgba(244,63,94,0.10)";
   287→      ctx.fillRect(x, narrTop + 2, Math.max(1, x2 - x), NARR_H - 4);
   288→    }
   289→
   290→    // ===== carril música =====
   291→    const musTop = RULER_H + NARR_H;
   292→    ctx.fillStyle = "#0d0d10";
   293→    ctx.fillRect(0, musTop, W, MUS_H);
   294→    ctx.strokeStyle = "#27272a";
   295→    ctx.beginPath();
   296→    ctx.moveTo(0, musTop + 0.5);
   297→    ctx.lineTo(W, musTop + 0.5);
   298→    ctx.stroke();
   299→    ctx.fillStyle = "#52525b";
   300→    ctx.font = "10px ui-monospace, SFMono-Regular, Menlo, monospace";
   301→    ctx.fillText("MÚSICA", 6, musTop + 6);
   302→
   303→    const yDe = (db: number) =>
   304→      musTop + MUS_H - clamp((db - DB_MIN) / (DB_MAX - DB_MIN), 0, 1) * MUS_H;
   305→
   306→    // rejilla dB
   307→    ctx.strokeStyle = "rgba(63,63,70,0.5)";
   308→    ctx.lineWidth = 1;
   309→    for (let db = DB_MIN; db <= DB_MAX; db += 12) {
   310→      const y = Math.round(yDe(db)) + 0.5;
   311→      ctx.beginPath();
   312→      ctx.moveTo(0, y);
   313→      ctx.lineTo(W, y);
   314→      ctx.stroke();
   315→    }
   316→
   317→    // bloques (con la vista previa del arrastre aplicada)
   318→    const bloquesVisibles = bloques.map(conArrastre);
   319→    for (const b of bloquesVisibles) {
   320→      if (b.fin + b.fadeOutSeg < t0 || b.inicio > t1) continue;
   321→      const pista = pistaMap.get(b.pistaId);
   322→      const color = pista
   323→        ? ANIMO_COLORES[pista.animo]
   324→        : "#a1a1aa";
   325→      const x = xDe(b.inicio);
   326→      const x2 = xDe(b.fin);
   327→      const w = Math.max(2, x2 - x);
   328→      const sel = selectedId === b.id;
   329→      ctx.globalAlpha = b.muted ? 0.25 : 0.8;
   330→      ctx.fillStyle = color;
   331→      ctx.fillRect(x, musTop + 14, w, MUS_H - 34);
   332→      ctx.globalAlpha = b.muted ? 0.4 : 1;
   333→      ctx.strokeStyle = sel ? "#ffffff" : color;
   334→      ctx.lineWidth = sel ? 1.8 : 1;
   335→      ctx.strokeRect(x + 0.5, musTop + 14.5, w - 1, MUS_H - 35);
   336→      ctx.globalAlpha = 1;
   337→
   338→      // costuras de bucle: dónde reinicia la pista dentro del bloque
   339→      const durPista = pista?.durSec ?? 0;
   340→      if (durPista > 1 && w > 24 && durPista * pps > 18) {
   341→        const offset = Math.max(0, b.offsetOrigen ?? 0);
   342→        ctx.strokeStyle = "rgba(9,9,11,0.55)";
   343→        ctx.lineWidth = 1;
   344→        ctx.setLineDash([2, 3]);
   345→        let ts = b.inicio + (durPista - (offset % durPista));
   346→        while (ts < b.fin - 0.2) {
   347→          const xs = xDe(ts);
   348→          if (xs > x + 12 && xs < x2 - 12) {
   349→            ctx.beginPath();
   350→            ctx.moveTo(xs + 0.5, musTop + 16);
   351→            ctx.lineTo(xs + 0.5, musTop + MUS_H - 22);
   352→            ctx.stroke();
   353→          }
   354→          ts += durPista;
   355→        }
   356→        ctx.setLineDash([]);
   357→      }
   358→
   359→      // asas de borde (estilo Camtasia): agarre para recortar/alargar
   360→      if (editable && onRecortar && w >= 8) {
   361→        const gw = Math.min(6, w / 4);
   362→        const activo =
   363→          arrastre?.modo === "recorte" && arrastre.id === b.id;
   364→        ctx.fillStyle = activo
   365→          ? "#fbbf24"
   366→          : sel
   367→            ? "rgba(251,191,36,0.9)"
   368→            : "rgba(255,255,255,0.55)";
   369→        ctx.fillRect(x, musTop + 14, gw, MUS_H - 34);
   370→        ctx.fillRect(x2 - gw, musTop + 14, gw, MUS_H - 34);
   371→      }
   372→
   373→      // cuñas de fundido
   374→      ctx.fillStyle = "rgba(9,9,11,0.55)";
   375→      if (b.fadeInSeg > 0.05 && b.inicio >= t0) {
   376→        const xf = Math.min(w, b.fadeInSeg * pps);
   377→        ctx.beginPath();
   378→        ctx.moveTo(x, musTop + 14);
   379→        ctx.lineTo(x + xf, musTop + 14);
   380→        ctx.lineTo(x, musTop + MUS_H - 20);
   381→        ctx.closePath();
   382→        ctx.fill();
   383→      }
   384→      if (b.fadeOutSeg > 0.05) {
   385→        const xf = Math.min(w, b.fadeOutSeg * pps);
   386→        ctx.beginPath();
   387→        ctx.moveTo(x2, musTop + 14);
   388→        ctx.lineTo(x2 - xf, musTop + 14);
   389→        ctx.lineTo(x2, musTop + MUS_H - 20);
   390→        ctx.closePath();
   391→        ctx.fill();
   392→      }
   393→
   394→      // etiqueta
   395→      if (w > 55 && pista) {
   396→        ctx.save();
   397→        ctx.beginPath();
   398→        ctx.rect(x + 2, musTop + 16, w - 4, 30);
   399→        ctx.clip();
   400→        ctx.fillStyle = "#09090b";
   401→        ctx.font = "bold 10px ui-sans-serif, system-ui, sans-serif";
   402→        ctx.fillText(pista.titulo, x + 6, musTop + 28);
   403→        ctx.font = "9px ui-monospace, monospace";
   404→        ctx.fillStyle = "rgba(255,255,255,0.75)";
   405→        ctx.fillText(
   406→          `${Math.round(b.fin - b.inicio)}s${
   407→            pista && buclesDe(b, pista.durSec) > 1
   408→              ? ` · bucle×${buclesDe(b, pista.durSec)}`
   409→              : ""
   410→          } · ${pista.bpm} BPM`,
   411→          x + 6,
   412→          musTop + 40
   413→        );
   414→        ctx.restore();
   415→      }
   416→      if (b.muted && w > 30) {
   417→        ctx.fillStyle = "rgba(228,228,231,0.85)";
   418→        ctx.font = "9px ui-sans-serif, system-ui, sans-serif";
   419→        ctx.fillText("SILENCIADO", x + 6, musTop + MUS_H - 26);
   420→      }
   421→    }
   422→
   423→    // envolvente de nivel (dB → y)
   424→    if (nivelDb && nivelDb.length > 0 && hopSec > 0) {
   425→      // zona de ducking (entre nivel y nivel+duck)
   426→      if (duckDb && duckDb.length === nivelDb.length) {
   427→        ctx.beginPath();
   428→        let empezo = false;
   429→        const n = nivelDb.length;
   430→        for (let i = 0; i < n; i++) {
   431→          const t = i * hopSec;
   432→          if (t < t0 - hopSec || t > t1 + hopSec) continue;
   433→          const x = xDe(t);
   434→          if (!empezo) {
   435→            ctx.moveTo(x, yDe(nivelDb[i]));
   436→            empezo = true;
   437→          } else {
   438→            ctx.lineTo(x, yDe(nivelDb[i]));
   439→          }
   440→        }
   441→        for (let i = n - 1; i >= 0; i--) {
   442→          const t = i * hopSec;
   443→          if (t < t0 - hopSec || t > t1 + hopSec) continue;
   444→          ctx.lineTo(xDe(t), yDe(nivelDb[i] + duckDb[i]));
   445→        }
   446→        ctx.closePath();
   447→        ctx.fillStyle = "rgba(244,63,94,0.14)";
   448→        ctx.fill();
   449→      }
   450→
   451→      // línea de nivel
   452→      ctx.beginPath();
   453→      let empezo2 = false;
   454→      for (let i = 0; i < nivelDb.length; i++) {
   455→        const t = i * hopSec;
   456→        if (t < t0 - hopSec || t > t1 + hopSec) continue;
   457→        const x = xDe(t);
   458→        const y = yDe(nivelDb[i]);
   459→        if (!empezo2) {
   460→          ctx.moveTo(x, y);
   461→          empezo2 = true;
   462→        } else {
   463→          ctx.lineTo(x, y);
   464→        }
   465→      }
   466→      ctx.strokeStyle = "rgba(245,158,11,0.95)";
   467→      ctx.lineWidth = 1.6;
   468→      ctx.stroke();
   469→    }
   470→
   471→    // playhead
   472→    const xPh = xDe(playheadSec);
   473→    if (xPh >= -2 && xPh <= W + 2) {
   474→      ctx.strokeStyle = "#fafafa";
   475→      ctx.lineWidth = 1;
   476→      ctx.beginPath();
   477→      ctx.moveTo(xPh + 0.5, 0);
   478→      ctx.lineTo(xPh + 0.5, H - 1);
   479→      ctx.stroke();
   480→      ctx.fillStyle = "#fafafa";
   481→      ctx.beginPath();
   482→      ctx.moveTo(xPh - 5, RULER_H - 8);
   483→      ctx.lineTo(xPh + 5, RULER_H - 8);
   484→      ctx.lineTo(xPh, RULER_H - 1);
   485→      ctx.closePath();
   486→      ctx.fill();
   487→    }
   488→
   489→    // hover
   490→    if (hover) {
   491→      ctx.strokeStyle = "rgba(250,250,250,0.25)";
   492→      ctx.setLineDash([3, 3]);
   493→      ctx.beginPath();
   494→      ctx.moveTo(hover.x + 0.5, RULER_H);
   495→      ctx.lineTo(hover.x + 0.5, H - 1);
   496→      ctx.stroke();
   497→      ctx.setLineDash([]);
   498→      ctx.fillStyle = "rgba(9,9,11,0.92)";
   499→      const texto = formatTimecode(hover.t);
   500→      ctx.font = "10px ui-monospace, monospace";
   501→      const tw = ctx.measureText(texto).width + 10;
   502→      const bx = clamp(hover.x - tw / 2, 2, W - tw - 2);
   503→      ctx.fillRect(bx, 2, tw, 16);
   504→      ctx.fillStyle = "#e4e4e7";
   505→      ctx.fillText(texto, bx + 5, 5);
   506→    }
   507→  }, [
   508→    ancho,
   509→    durationSec,
   510→    vistaParaDibujo,
   511→    analysis,
   512→    silences,
   513→    bloques,
   514→    arrastre,
   515→    pistaMap,
   516→    selectedId,
   517→    nivelDb,
   518→    duckDb,
   519→    hopSec,

   533→    (v: View): View => {
   534→      const pps = clamp(v.pps, 0.05, MAX_PPS);

   541→  const zoom = useCallback(
   542→    (factor: number, anclaX: number) => {
   543→      setVista((v) => {
   544→        const base = v ?? { start: 0, pps: ancho / Math.max(1, durationSec) };
   545→        const tAncla = base.start + anclaX / base.pps;
   546→        const nueva = { pps: base.pps * factor, start: 0 };
   547→        nueva.start = tAncla - anclaX / nueva.pps;
   548→        return clampVista(nueva);
   549→      });
   550→    },
   551→    [ancho, durationSec, clampVista]

   554→  useEffect(() => {

   563→    return () => el.removeEventListener("wheel", onWheel);

   566→  const hitBloque = useCallback(
   567→    (x: number, y: number): BloqueMusica | null => {
   568→      const musTop = RULER_H + NARR_H;
   569→      if (y < musTop + 14 || y > musTop + MUS_H - 20) return null;

   573→      return (
   574→        lista.find((b) => t >= b.inicio && t <= b.fin) ??
   575→        lista.find((b) => t >= b.inicio - 0.3 && t <= b.fin + 0.3) ??
   576→        null
   577→      );
   578→    },

   582→  /** ¿El puntero está sobre el borde (asas) de un bloque? */
   583→  const hitBorde = useCallback(
   584→    (x: number, y: number): { id: string; lado: LadoRecorte } | null => {
   585→      if (!editable || !onRecortar) return null;
   586→      const musTop = RULER_H + NARR_H;
   587→      if (y < musTop + 14 || y > musTop + MUS_H - 20) return null;
   588→      const { start: t0, pps } = vistaEfectiva;
   589→      const t = t0 + x / pps;
   590→      const umbral = clamp(EDGE_PX / pps, 0.12, 1.2);
   591→      let mejor: { id: string; lado: LadoRecorte } | null = null;
   592→      let mejorDt = Infinity;
   593→      for (const b of bloques) {
   594→        for (const lado of ["inicio", "fin"] as const) {
   595→          const borde = lado === "inicio" ? b.inicio : b.fin;
   596→          const dt = Math.abs(t - borde);
   597→          if (dt < umbral && dt < mejorDt) {
   598→            mejor = { id: b.id, lado };
   599→            mejorDt = dt;
   600→          }
   601→        }
   602→      }
   603→      return mejor;
   604→    },

   614→  const onPointerDown = (e: React.PointerEvent<HTMLCanvasElement>) => {
   615→    const el = canvasRef.current;
   616→    if (!el) return;
   617→    const rect = el.getBoundingClientRect();
   618→    const x = e.clientX - rect.left;
   619→    const y = e.clientY - rect.top;
   620→    el.setPointerCapture(e.pointerId);
   621→    const { start: t0, pps } = vistaEfectiva;
   622→    const t = t0 + x / pps;
   623→
   624→    // arrastre de bloque (solo editable): primero los BORDES (recorte estilo
   625→    // Camtasia), luego el cuerpo (mover)
   626→    if (editable) {
   627→      const borde = hitBorde(x, y);
   628→      if (borde) {
   629→        const b = bloques.find((x2) => x2.id === borde.id);
   630→        if (b) {
   631→          onSelect?.(b.id);
   632→          enArrastre.current = true;
   633→          setArrastre({
   634→            modo: "recorte",
   635→            id: b.id,
   636→            lado: borde.lado,
   637→            valorOriginal: borde.lado === "inicio" ? b.inicio : b.fin,
   638→            valor: borde.lado === "inicio" ? b.inicio : b.fin,
   639→          });
   640→          pressRef.current = { x, y, t, tiempo: performance.now() };
   641→          ponerCursor("ew-resize");
   642→          return;
   643→        }
   644→      }
   645→      const b = hitBloque(x, y);
   646→      if (b) {
   647→        onSelect?.(b.id);
   648→        enArrastre.current = true;
   649→        setArrastre({
   650→          modo: "mover",
   651→          id: b.id,
   652→          inicioOriginal: b.inicio,
   653→          deltaSec: 0,
   654→        });
   655→        pressRef.current = { x, y, t, tiempo: performance.now() };
   656→        ponerCursor("grabbing");
   657→        return;
   658→      }
   659→    }
   660→
   661→    // pan desde la regla
   662→    if (y <= RULER_H) {
   663→      panRef.current = { x, start: t0 };
   664→      pressRef.current = null;
   665→      return;
   666→    }
   667→
   668→    pressRef.current = { x, y, t, tiempo: performance.now() };
   669→    if (!editable) {
   670→      // clic en el carril de voz = scrub
   671→      onSeek(clamp(t, 0, durationSec));
   672→    }

   676→    const el = canvasRef.current;
   677→    if (!el) return;
   678→    const rect = el.getBoundingClientRect();
   679→    const x = e.clientX - rect.left;
   680→    const y = e.clientY - rect.top;
   681→    const { start: t0, pps } = vistaEfectiva;
   682→    const t = t0 + x / pps;
   683→    if (y > RULER_H) setHover({ x, t });
   684→    else setHover(null);
   685→
   686→    // cursor según la zona (borde = recorte, cuerpo = mover)
   687→    if (!enArrastre.current && !panRef.current) {
   688→      if (editable) {
   689→        const borde = hitBorde(x, y);
   690→        ponerCursor(
   691→          borde
   692→            ? "ew-resize"
   693→            : hitBloque(x, y)
   694→              ? "grab"
   695→              : "crosshair"
   696→        );
   697→      } else {
   698→        ponerCursor("crosshair");
   699→      }
   700→    }
   701→
   702→    if (panRef.current) {
   703→      const dt = (x - panRef.current.x) / pps;
   704→      setVista(clampVista({ pps, start: panRef.current.start - dt }));
   705→      return;
   706→    }
   707→
   708→    if (enArrastre.current && arrastre) {
   709→      if (arrastre.modo === "recorte") {
   710→        setArrastre({ ...arrastre, valor: t });
   711→      } else {
   712→        const delta = (x - (pressRef.current?.x ?? x)) / pps;
   713→        setArrastre({ ...arrastre, deltaSec: Math.round(delta * 10) / 10 });
   714→      }
   715→    }
   716→  };

   718→  const onPointerUp = (e: React.PointerEvent<HTMLCanvasElement>) => {
   719→    const el = canvasRef.current;
   720→    if (!el) return;
   721→    try {
   722→      el.releasePointerCapture(e.pointerId);
   723→    } catch {
   724→      /* sin captura */
   725→    }

   736→          Math.abs(arrastre.valor - arrastre.valorOriginal) > 0.05;
   737→        if (movio && onRecortar && editable) {

   750→        onMover(arrastre.id, Math.round(nuevoInicio * 10) / 10);
   751→      } else if (!movio) {
   752→        // clic simple sobre el bloque → seek al inicio del bloque

   754→      }
   755→      return;

   762→        // clic simple en la regla → seek

   765→        onSeek(clamp(t, 0, durationSec));

   768→    }

   783→        }
   784→      }
   785→      pressRef.current = null;
